<script>
// CONFIGURACIÓN DEL CONTRATO (Debe modificarse con los datos reales del
despliegue)
// Public Key del programa Anchor desplegado
const PROGRAM_ID_STR = "GradingSystem1111111111111111111111111111111";
// Public Key institucional del profesor (utilizada como semilla fija en este ejemplo)
const TEACHER_PUBKEY_STR =
"ProfXxxxxx11111111111111111111111111111111";
// Endpoint RPC Público de Devnet
const RPC_ENDPOINT = "https://api.devnet.solana.com";
// Elementos del DOM
const searchForm = document.getElementById('searchForm');
const searchBtn = document.getElementById('searchBtn');
const statusContainer = document.getElementById('statusContainer');
const loadingState = document.getElementById('loadingState');
const resultState = document.getElementById('resultState');
const alertState = document.getElementById('alertState');
const alertIconContainer = document.getElementById('alertIconContainer');
const alertIcon = document.getElementById('alertIcon');
const alertTitle = document.getElementById('alertTitle');
const alertDescription = document.getElementById('alertDescription');
// Elementos de la Tarjeta de Resultados
const gradeDisplay = document.getElementById('gradeDisplay');
const metaStudent = document.getElementById('metaStudent');
const metaSubject = document.getElementById('metaSubject');
const metaDate = document.getElementById('metaDate');
const metaTeacher = document.getElementById('metaTeacher');
// Inicializar Conexión RPC de Solana sin necesidad de Wallet
const connection = new solanaWeb3.Connection(RPC_ENDPOINT, 'confirmed');
const programId = new solanaWeb3.PublicKey(PROGRAM_ID_STR);
const teacherPubkey = new solanaWeb3.PublicKey(TEACHER_PUBKEY_STR);
searchForm.addEventListener('submit', async (e) => {
e.preventDefault();
// Obtener y normalizar inputs del estudiante
const studentCode =
document.getElementById('studentCode').value.trim().toUpperCase();
const subjectCode =
document.getElementById('subjectCode').value.trim().toUpperCase();
// Activar Estado de Carga
setLoading(true);
try {
// 1. Derivación de la Cuenta PDA bajo las reglas de Anchor
// Semillas: [b"grade", pubkey_del_profesor, codigo_estudiante,
codigo_asignatura]
const [pdaAddress, _bump] = await solanaWeb3.PublicKey.findProgramAddress(
[
Array.from(new TextEncoder().encode("grade")),
teacherPubkey.toBuffer(),
Array.from(new TextEncoder().encode(studentCode)),
Array.from(new TextEncoder().encode(subjectCode))
],
programId
);
// 2. Fetch de la información de la cuenta en la red de Solana
const accountInfo = await connection.getAccountInfo(pdaAddress);
if (accountInfo === null) {
// La cuenta no existe (Calificación no encontrada)
showAlert(
"warning",
"Calificación no encontrada",
"Es posible que el profesor aún no la haya firmado o los códigos ingresados
sean incorrectos."
);
} else {
// 3. Deserialización de los datos (Borsh Emulation)
// Estructura de Anchor esperada:
// [8 Bytes de Discriminador de Anchor] + [2 Bytes para u16 de la Nota] + [8
Bytes para i64 de Timestamp Unix]
const data = accountInfo.data;
// Verificación de tamaño seguro mínimo de datos devueltos por la blockchain
if(data.length >= 18) {
// Leer un entero sin signo de 16 bits después del discriminador de Anchor
(offset 8)
const rawGrade = data.readUInt16LE ? data.readUInt16LE(8) : new
DataView(data.buffer, data.byteOffset, data.byteLength).getUint16(8, true);
// Convertir los dos decimales implícitos (ej. 9550 -> 95.50)
const finalGrade = (rawGrade / 100).toFixed(2);
// Leer el Timestamp Unix de 64 bits (offset 10)
// Para simplificar Vanilla JS, leemos los 32 bits inferiores (relevantes para
timestamps actuales)
const timestampOffset = 10;
const view = new DataView(data.buffer, data.byteOffset, data.byteLength);
const unixTimestamp = view.getUint32(timestampOffset, true); // Little Endian
// Conversión de fecha DD/MM/AAAA
const dateObj = new Date(unixTimestamp * 1000);
const formattedDate = dateObj.toLocaleDateString('es-ES', {
day: '2-digit',
month: '2-digit',
year: 'numeric'
});
// Renderizar Datos con éxito
gradeDisplay.innerText = finalGrade;
metaStudent.innerText = studentCode;
metaSubject.innerText = subjectCode;
metaDate.innerText = formattedDate;
metaTeacher.innerText = teacherPubkey.toBase58();
showResult();
} else {
throw new Error("Estructura de datos corrupta o inválida.");
}
}
} catch (error) {
console.error("Solana RPC Error:", error);
showAlert(
"error",
"Error al conectar con la blockchain",
"No se pudo establecer comunicación con el nodo RPC de Solana Devnet.
Revisa tu conexión a internet e inténtalo de nuevo."
);
} finally {
setLoading(false);
}
});
// Controladores de UI y Estados de la UX
function setLoading(isLoading) {
statusContainer.classList.remove('hidden');
if (isLoading) {
loadingState.classList.remove('hidden');
resultState.classList.add('hidden');
alertState.classList.add('hidden');
searchBtn.disabled = true;
searchBtn.classList.add('opacity-70', 'cursor-not-allowed');
} else {
loadingState.classList.add('hidden');
searchBtn.disabled = false;
searchBtn.classList.remove('opacity-70', 'cursor-not-allowed');
}
}
function showResult() {
resultState.classList.remove('hidden');
alertState.classList.add('hidden');
}
function showAlert(type, title, description) {
resultState.classList.add('hidden');
alertState.classList.remove('hidden');
alertTitle.innerText = title;
alertDescription.innerText = description;
if (type === "warning") {
// Estilo Amarillo/Dorado institucional para avisos
alertState.className = "border-amber-200 bg-amber-50 text-amber-900 rounded-
xl p-6 flex flex-col sm:flex-row items-center gap-4 transition-all duration-300";
alertIconContainer.className = "p-3 rounded-full bg-amber-100 text-amber-600";
alertIcon.innerHTML = `<path stroke-linecap="round" stroke-linejoin="round"
stroke-width="2" d="M12 9v2m0 4h.01m-6.938 4h13.856c1.54 0 2.502-1.667 1.732-
3L13.732 4c-.77-1.333-2.694-1.333-3.464 0L3.34 16c-.77 1.333.192 3 1.732 3z"/>`;
} else {
// Estilo Rojo para errores de red o código
alertState.className = "border-red-200 bg-red-50 text-red-950 rounded-xl p-6 flex
flex-col sm:flex-row items-center gap-4 transition-all duration-300";
alertIconContainer.className = "p-3 rounded-full bg-red-100 text-red-600";
alertIcon.innerHTML = `<path stroke-linecap="round" stroke-linejoin="round"
stroke-width="2" d="M12 8v4m0 4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z"/>`;
}
}
</script>
